		On Install:
		"
			if ! grep -qF "$(which zsh)" /etc/shells ; then
         			echo "$(which zsh)" | sudo tee -a /etc/shells
			fi
			
			sudo chsh -s $(which zsh)
		"

		Sets zsh to be the default shell, making sure it's
		in the list of authorized shells.
